#### 代码示例

Bellman-Ford 算法：

```cpp
/*
以下实现 Bellman-Ford 算法
n：顶点数量，m：边数，edges[1..m]：所有有向边
dis：source 到各点的最短距离
函数在存在 source 可达的负环时返回 false
*/

const ll LL_INF = 1LL << 60;
const int MAXN = 2e5 + 5;
const int MAXM = 2e5 + 5;

struct edge{
    int u, v;
    ll w;
} edges[MAXM];

int n, m;
ll dis[MAXN];

bool bellmanFord(int source){
    fill(dis + 1, dis + n + 1, LL_INF);
    dis[source] = 0;

    // 不含负环的最短简单路径至多经过 n - 1 条边
    for (int k = 1; k <= n - 1; k++){
        bool flag = false;
        for (int i = 1; i <= m; i++){
            edge e = edges[i];
            if (dis[e.u] == LL_INF) continue;
            if (dis[e.u] + e.w < dis[e.v]){
                dis[e.v] = dis[e.u] + e.w;
                flag = true;
            }
        }
        if (!flag) break;
    }

    // 再松弛一次仍能变短，说明存在源点可达的负权环
    for (int i = 1; i <= m; i++){
        edge e = edges[i];
        if (dis[e.u] != LL_INF && dis[e.u] + e.w < dis[e.v]) return false;
    }
    return true;
}
```

Dijkstra 算法：

```cpp
/*
以下实现堆优化 Dijkstra 算法
G：非负权有向图，source：源点
dis：source 到各点的最短距离，不可达点为 LL_INF
*/

struct edge{
    int v;
    ll w;
};

vector<edge> G[MAXN];
ll dis[MAXN];
int n;

void dijkstra(int source){
    typedef pair<ll, int> state;
    priority_queue<state, vector<state>, greater<state> > q;
    fill(dis + 1, dis + n + 1, LL_INF);
    dis[source] = 0;
    q.push(state{0, source});

    while (!q.empty()){
        state now = q.top();
        q.pop();
        ll d = now.first;
        int u = now.second;
        // 跳过距离更新后遗留在堆中的旧状态
        if (d != dis[u]) continue;

        for (edge e: G[u]){
            int v = e.v;
            if (dis[u] + e.w < dis[v]){
                dis[v] = dis[u] + e.w;
                q.push(state{dis[v], v});
            }
        }
    }
}
```

SPFA 算法：

```cpp
/*
以下实现 SPFA 算法
G：有向图，source：源点，dis：source 到各点的最短距离
inQue：顶点是否在队列中；图中不存在 source 可达的负环
*/

bool inQue[MAXN];

void spfa(int source){
    fill(dis + 1, dis + n + 1, LL_INF);
    fill(inQue + 1, inQue + n + 1, false);
    queue<int> q;
    dis[source] = 0;
    q.push(source);
    inQue[source] = true;

    // 反复取出距离变小的节点，松弛其所有出边
    while (!q.empty()){
        int u = q.front();
        q.pop();
        inQue[u] = false;
        for (edge e: G[u]){
            int v = e.v;
            if (dis[u] + e.w >= dis[v]) continue;
            dis[v] = dis[u] + e.w;
            if (!inQue[v]){
                q.push(v);
                inQue[v] = true;
            }
        }
    }
}
```

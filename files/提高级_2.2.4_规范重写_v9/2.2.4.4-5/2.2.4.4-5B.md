#### 代码示例

```cpp
/*
函数 radixSort 可完成对 32 位无符号整数数组 a 的基数排序
a：待排序数组，下标从 1 开始
temp：存放一趟计数排序的结果
cnt：存放关键字的计数及其前缀和
n：待排序元素的数量，BASE：基数 65536
*/

const int BASE = 1 << 16;
int n, cnt[BASE];
uint32_t a[MAXN], temp[MAXN];

void countingSort(uint32_t source[], uint32_t target[], int shift){
    fill(cnt, cnt + BASE, 0);
    // 统计当前 16 位关键字的出现次数
    for (int i = 1; i <= n; i++){
        int digit = int((source[i] >> shift) & uint32_t(BASE - 1));
        cnt[digit]++;
    }
    for (int i = 1; i < BASE; i++){
        cnt[i] += cnt[i - 1];
    }
    // 逆序放置，使关键字相同的元素保持原有相对顺序
    for (int i = n; i >= 1; i--){
        int digit = int((source[i] >> shift) & uint32_t(BASE - 1));
        target[cnt[digit]] = source[i];
        cnt[digit]--;
    }
}

void radixSort(){
    // 先按低 16 位稳定排序，再按高 16 位稳定排序
    countingSort(a, temp, 0);
    countingSort(temp, a, 16);
}
```

#### 代码示例

单调队列数据结构：

```cpp
/*
以下使用数组实现维护最大值的单调队列
qVal/qTm：按值递减、按严格递增的时间戳存储队列元素
head/tail：队首和队尾下标，head > tail 表示空；调用 queryQueue 时队列非空
*/

int qVal[MAXN], qTm[MAXN];
int head = 1, tail = 0;

void pushQueue(int val, int tm){
    // 队尾较小元素不可能再成为后续窗口最大值
    while (head <= tail && qVal[tail] <= val) tail--;
    qVal[++tail] = val;
    qTm[tail] = tm;
}

int queryQueue(int tm){
    // 删除时间戳小于下界的过期元素
    while (head <= tail && qTm[head] < tm) head++;
    return qVal[head];
}
```

使用单调队列求滑动窗口最大值：

```cpp
/*
函数 slidingWindowMax 使用单调队列求长度为 k 的滑动窗口最大值
a[1..n]：输入序列，ans：各窗口最大值
函数返回窗口数量，k 非法时返回 0
*/

int slidingWindowMax(int a[], int n, int k, int ans[]){
    if (k < 1 || k > n) return 0;
    head = 1;
    tail = 0;
    int cnt = 0;
    // 从第 k 个元素起，每次产生一个完整窗口
    for (int i = 1; i <= n; i++){
        pushQueue(a[i], i);
        if (i >= k) ans[++cnt] = queryQueue(i - k + 1);
    }
    return cnt;
}
```

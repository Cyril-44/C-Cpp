#### 代码示例

```cpp
/*
函数 countWays 使用状态压缩动态规划求国王放置方案数
n：棋盘边长，k：国王数量，1 <= n <= 9
state：所有行内没有相邻国王的合法状态
kingCnt：各行状态的国王数量
dp[row][id][placed]：前 row 行放置 placed 个国王且末行状态为 id 的方案数
*/

const int MAXBOARD = 9;
const int MAXSTATE = 1 << MAXBOARD;
const int MAXKING = MAXBOARD * MAXBOARD;

int n, k, state[MAXSTATE + 1], kingCnt[MAXSTATE + 1];
int stateCnt, emptyId;
ll dp[MAXBOARD + 1][MAXSTATE + 1][MAXKING + 1];

bool isCompatible(int lower, int upper){
    // 排除正上方、左上方和右上方的相邻国王
    return (upper & lower) == 0 &&
        (upper & (lower << 1)) == 0 &&
        (upper & (lower >> 1)) == 0;
}

void init(){
    stateCnt = 0;
    int limit = 1 << n;
    // 相邻二进制位不能同时为 1
    for (int mask = 0; mask < limit; mask++){
        if ((mask & (mask << 1)) != 0) continue;
        stateCnt++;
        state[stateCnt] = mask;
        kingCnt[stateCnt] = __builtin_popcount(mask);
        if (mask == 0) emptyId = stateCnt;
    }
}

ll countWays(){
    if (k < 0 || k > n * n) return 0;
    init();
    memset(dp, 0, sizeof(dp));
    dp[0][emptyId][0] = 1;

    // 枚举当前行和上一行的合法状态
    for (int row = 1; row <= n; row++){
        for (int i = 1; i <= stateCnt; i++){ // 当前行状态
            int cnt = kingCnt[i];
            for (int j = 1; j <= stateCnt; j++){ // 上一行状态
                if (!isCompatible(state[i], state[j])) continue;
                for (int placed = cnt; placed <= k; placed++){
                    dp[row][i][placed] += dp[row - 1][j][placed - cnt];
                }
            }
        }
    }

    ll ans = 0;
    for (int id = 1; id <= stateCnt; id++){
        ans += dp[n][id][k];
    }
    return ans;
}
```

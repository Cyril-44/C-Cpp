#### 代码示例

```cpp
/*
t：模式串，s：文本串
n：文本串长度，m：模式串长度，m >= 1
fail：模式串的失配指针
函数 kmp 返回模式串出现次数，允许重叠匹配
*/

int n, m, fail[MAXN];
char s[MAXN], t[MAXN];

void init(){
    fail[1] = 0;
    int k = 0;
    for (int i = 2; i <= m; i++){
        // 当前字符失配时，不移动 i，只缩短已匹配的前后缀
        while (k > 0 && t[k + 1] != t[i]){
            k = fail[k];
        }
        if (t[k + 1] == t[i]) k++;
        fail[i] = k;
    }
}

int kmp(){
    init();
    int cnt = 0;
    int k = 0;
    for (int i = 1; i <= n; i++){
        // 沿 fail 链寻找仍可能与 s[i] 匹配的前缀
        while (k > 0 && t[k + 1] != s[i]){
            k = fail[k];
        }
        if (t[k + 1] == s[i]) k++;
        if (k == m){
            cnt++;
            // 回退到最长相等前后缀，以继续统计重叠出现
            k = fail[k];
        }
    }
    return cnt;
}
```

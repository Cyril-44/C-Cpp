#### 代码示例

```cpp
/*
函数 heapSort 使用优先队列将数组 a 按从小到大排序
a：待排序数组，下标从 1 开始，排序结果仍写回 a
n：待排序元素的数量
*/
int n, a[MAXN];

void heapSort(){
    // 使用 greater<int> 将默认大根堆改为小根堆
    priority_queue<int, vector<int>, greater<int> > q;
    for (int i = 1; i <= n; i++) q.push(a[i]);

    // 最小值依次写回数组，最终得到升序数组
    for (int i = 1; i <= n; i++){
        a[i] = q.top();
        q.pop();
    }
}
```

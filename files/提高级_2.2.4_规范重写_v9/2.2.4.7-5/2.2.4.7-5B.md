#### 代码示例

Floyd-Warshall 算法求全源最短路：

```cpp
/*
以下为 Floyd-Warshall 算法求全源最短路
dis：任意两点之间的距离，不可达时为 LL_INF
n：顶点数量，图中不存在负环
*/

const int MAXN = 1e3 + 5;
const ll LL_INF = 1LL << 60;
ll dis[MAXN][MAXN];
int n;

void floyd(){
    // 第 k 轮允许节点 k 作为中间点
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++){
                if (dis[i][k] == LL_INF || dis[k][j] == LL_INF) continue;
                dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j]);
            }
}
```

Floyd-Warshall 算法求有向图传递闭包：

```cpp
/*
以下为 Floyd-Warshall 算法求有向图传递闭包
reach[i][j]：顶点 i 是否能够到达顶点 j
n：顶点数量，满足 n < MAXN
*/

bitset<MAXN> reach[MAXN];

void floydClosure(){
    // 若 i 能到达 k，则并入 k 能到达的所有节点
    for (int i = 1; i <= n; i++) reach[i][i] = true;
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            if (reach[i][k]) reach[i] |= reach[k];
}
```

#### 代码示例

```cpp
/*
以下代码求非负权有向图的严格次短路
d1：最短路长度，d2：严格次短路长度
不存在时距离为 LL_INF
*/

const ll LL_INF = 1LL << 60;

struct edge{
    int v;
    ll w;
};

vector<edge> G[MAXN];
ll d1[MAXN], d2[MAXN];
int n;

void secondShortest(int source){
    typedef pair<ll, int> state;
    priority_queue<state, vector<state>, greater<state> > q;
    fill(d1 + 1, d1 + n + 1, LL_INF);
    fill(d2 + 1, d2 + n + 1, LL_INF);
    q.push(state{0, source});

    // 每个节点只扩展最小和严格次小的两个不同距离
    while (!q.empty()){
        state now = q.top();
        q.pop();
        ll d = now.first;
        int u = now.second;
        if (d < d1[u]){
            d1[u] = d;
        } else if (d1[u] < d && d < d2[u]){
            d2[u] = d;
        } else {
            // 跳过重复距离和多余的第三种距离
            continue;
        }
        for (edge e: G[u]){
            if (e.w > LL_INF - d) continue;
            q.push(state{d + e.w, e.v});
        }
    }
}
```

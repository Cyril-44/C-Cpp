#### 代码示例

```cpp
/*
以下代码求无向图的严格次小生成树
n：顶点数量，m：边数，edges[1..m]：所有无向边
T：最小生成树邻接表，元素为 (相邻节点, 边权)
wt[u][i]：向上 2^i 条边中的最大值与严格次大值
调用前已求出 mstWeight，并用 inTree 标记树边；无解时返回 -1
*/

const int MAXLOG = 20;
const int MAXN = 2e5 + 5;
const int MAXM = 2e5 + 5;
const ll LL_INF = 1LL << 60;
const ll NEG_INF = -LL_INF;

struct graphEdge{
    int u, v;
    ll w;
    bool inTree;
} edges[MAXM];

struct max2{
    ll fst, sec;
};

vector<pair<int, ll> > T[MAXN];
int n, m;
ll mstWeight;
int dep[MAXN], anc[MAXN][MAXLOG];
max2 wt[MAXN][MAXLOG];

max2 combine(max2 a, max2 b){
    // 先确定最大值，再从剩余候选值中取严格次大值
    if (a.fst < b.fst) swap(a, b);
    if (a.fst == b.fst)
        a.sec = max(a.sec, b.sec);
    else
        a.sec = max(a.sec, b.fst);
    return a;
}

void dfs(int u, int parent, ll w){
    dep[u] = dep[parent] + 1;
    anc[u][0] = parent;
    wt[u][0] = max2{w, NEG_INF};
    // 合并两个 2^(i-1) 段，得到向上 2^i 条边的信息
    for (int i = 1; i < MAXLOG; i++){
        int p = anc[u][i - 1];
        anc[u][i] = anc[p][i - 1];
        wt[u][i] = combine(wt[u][i - 1], wt[p][i - 1]);
    }
    for (pair<int, ll> e: T[u]){
        int v = e.first;
        if (v == parent) continue;
        dfs(v, u, e.second);
    }
}

void init(){
    dep[0] = 0;
    for (int i = 0; i < MAXLOG; i++) wt[0][i] = max2{NEG_INF, NEG_INF};
    dfs(1, 0, NEG_INF);
}

max2 query(int u, int v){
    max2 res{NEG_INF, NEG_INF};
    if (dep[u] < dep[v]) swap(u, v);
    // 倍增维护路径上最大的两个不同边权
    for (int i = MAXLOG - 1; i >= 0; i--){
        if (dep[anc[u][i]] >= dep[v]){
            res = combine(res, wt[u][i]);
            u = anc[u][i];
        }
    }
    if (u == v) return res;

    for (int i = MAXLOG - 1; i >= 0; i--){
        if (anc[u][i] != anc[v][i]){
            res = combine(res, wt[u][i]);
            res = combine(res, wt[v][i]);
            u = anc[u][i];
            v = anc[v][i];
        }
    }
    res = combine(res, wt[u][0]);
    res = combine(res, wt[v][0]);
    return res;
}

ll secondMst(){
    init();

    ll ans = LL_INF;
    // 加入非树边后，删去环上一条边得到候选生成树
    for (int i = 1; i <= m; i++){
        if (edges[i].inTree || edges[i].u == edges[i].v) continue;
        max2 x = query(edges[i].u, edges[i].v);
        if (edges[i].w > x.fst)
            ans = min(ans, mstWeight + edges[i].w - x.fst);
        else if (edges[i].w == x.fst && x.sec != NEG_INF)
            ans = min(ans, mstWeight + edges[i].w - x.sec);
    }
    return ans == LL_INF ? -1 : ans;
}
```

#### 代码示例

```cpp
/*
函数 bucketSort 对小整数值域数组 a 进行从小到大排序
a：待排序数组，下标从 1 开始，元素取值范围为 [1, m]
bucket[x]：值 x 的出现次数
n：待排序元素的数量，m：值域上限，1 <= m < MAXM
*/

const int MAXN = 1e5 + 5;
const int MAXM = 1e5 + 5;
int n, m, a[MAXN], bucket[MAXM];

void bucketSort(){
    fill(bucket + 1, bucket + m + 1, 0);
    // 每个可能的值对应一个桶
    for (int i = 1; i <= n; i++) bucket[a[i]]++;

    // 按值从小到大取出各桶中的元素
    int pos = 0;
    for (int value = 1; value <= m; value++){
        while (bucket[value] > 0){
            a[++pos] = value;
            bucket[value]--;
        }
    }
}
```

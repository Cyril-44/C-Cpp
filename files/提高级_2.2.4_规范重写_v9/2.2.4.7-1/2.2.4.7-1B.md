#### 代码示例

Prim 算法：

```cpp
/*
以下实现 Prim 算法
函数 prim 在图连通时返回最小生成树边权和，否则返回 -1
G[u][v]：u、v 间的最小边权，无边时为 LL_INF
dis：各点到当前生成树的最小边权
vis：各顶点是否已经加入生成树
*/

const int MAXN = 1e3 + 5;
const ll LL_INF = 1LL << 60;
ll G[MAXN][MAXN], dis[MAXN];
int n;
bool vis[MAXN];

ll prim(){
    // 从顶点 1 开始扩展生成树
    fill(dis + 1, dis + n + 1, LL_INF);
    fill(vis + 1, vis + n + 1, false);
    dis[1] = 0;
    ll ans = 0;
    for (int k = 1; k <= n; k++){
        // 选择树外 dis 最小的顶点
        int u = 0;
        for (int i = 1; i <= n; i++)
            if (!vis[i] && (u == 0 || dis[i] < dis[u])) u = i;
        if (u == 0 || dis[u] == LL_INF) return -1;
        vis[u] = true;
        ans += dis[u];
        // 用新加入顶点的边更新其余顶点
        for (int v = 1; v <= n; v++)
            if (!vis[v]) dis[v] = min(dis[v], G[u][v]);
    }
    return ans;
}
```

Kruskal 算法：

```cpp
/*
以下基于并查集实现 Kruskal 算法
n：顶点数量，m：边数
edges[1..m]：存储所有无向边，fa、sz：并查集父节点和集合大小
函数 kruskal 在图连通时返回最小生成树边权和，否则返回 -1
*/

struct edge{
    int u, v, w;
};

const int MAXN = 2e5 + 5;
const int MAXM = 2e5 + 5;
edge edges[MAXM];

int n, m, fa[MAXN], sz[MAXN];

bool cmp(const edge &a, const edge &b){
    return a.w < b.w;
}

int findRoot(int x){
    // 路径压缩
    if (fa[x] == x) return x;
    fa[x] = findRoot(fa[x]);
    return fa[x];
}

ll kruskal(){
    for (int i = 1; i <= n; i++){
        fa[i] = i;
        sz[i] = 1;
    }
    sort(edges + 1, edges + m + 1, cmp);

    ll ans = 0;
    int cnt = 0;
    // 按边权从小到大合并不同连通块
    for (int i = 1; i <= m; i++){
        int x = findRoot(edges[i].u);
        int y = findRoot(edges[i].v);
        if (x == y) continue;
        // 小集合并入大集合
        if (sz[x] < sz[y]) swap(x, y);
        fa[y] = x;
        sz[x] += sz[y];
        ans += ll(edges[i].w);
        cnt++;
    }
    if (cnt != n - 1) return -1;
    return ans;
}
```

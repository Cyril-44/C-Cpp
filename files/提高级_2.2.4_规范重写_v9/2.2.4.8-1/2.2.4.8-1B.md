#### 代码示例

```cpp
/*
以下使用树形动态规划求树的最大权独立集
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
weight：节点权值
dp[u][0/1]：不选/选择 u 时，以 u 为根的子树最大权值和
调用 dfs(root, 0) 后，max(dp[root][0], dp[root][1]) 为答案
*/

vector<int> T[MAXN];
ll weight[MAXN], dp[MAXN][2];
int n;

void dfs(int u, int parent){
    dp[u][0] = 0;
    dp[u][1] = weight[u];
    for (int v: T[u]){
        if (v == parent) continue;
        dfs(v, u);
        // u 不选时子节点可选可不选；u 选时子节点不能选
        dp[u][0] += max(dp[v][0], dp[v][1]);
        dp[u][1] += dp[v][0];
    }
}

```

建议将 deg 计算放到 topoSort() 中去。

#### 代码示例

```cpp
/*
以下使用 Kahn 算法求有向图的拓扑序
G：有向图邻接表，点编号 1~n
topoSort 返回拓扑序；图中有环时返回空序列
deg：每个节点的入度
*/

vector<int> G[MAXN];
int n;

vector<int> topoSort(){
    queue<int> q;
    vector<int> ans, deg(n+1);
    ans.reserve(n);
    for (int u = 1; u <= n; u++)
        for (int v : G[u])
            deg[v]++;
    for (int u = 1; u <= n; u++){
        if (deg[u] == 0) q.push(u);
    }

    while (!q.empty()){
        int u = q.front();
        q.pop();
        ans.push_back(u);
        // 删除 u 的所有出边，新产生的零入度节点随后入队
        for (int v : G[u]){
            deg[v]--;
            if (deg[v] == 0) q.push(v);
        }
    }
    // 有节点未被取出，等价于图中仍存在有向环
    if (int(ans.size()) != n) ans.clear();
    return ans;
}
```

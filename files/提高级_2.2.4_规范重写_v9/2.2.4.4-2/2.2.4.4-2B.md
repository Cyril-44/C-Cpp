#### 代码示例

```cpp
/*
函数 quickSort 可完成对数组指定闭区间内元素的快速排序
a：待排序数组，下标从 1 开始
l、r：当前待排序区间的左右端点
*/

int a[MAXN];
mt19937 rng(unsigned(time(0)));

void quickSort(int l, int r){
    if (l >= r) return;
    int x = a[l + int(rng() % (r - l + 1))];
    int lt = l, i = l, gt = r;
    // [l,lt-1]<x，[lt,i-1]=x，[i,gt] 待处理，[gt+1,r]>x
    while (i <= gt){
        if (a[i] < x){
            swap(a[lt++], a[i++]);
        } else if (a[i] > x){
            // 换来的 a[i] 尚未判断，因此这里只移动 gt
            swap(a[i], a[gt--]);
        } else {
            i++;
        }
    }
    // 等于 x 的中间段已有序，只递归两侧
    quickSort(l, lt - 1);
    quickSort(gt + 1, r);
}
```

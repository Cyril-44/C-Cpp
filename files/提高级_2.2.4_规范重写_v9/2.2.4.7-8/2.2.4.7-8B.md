#### 代码示例

```cpp
/*
以下采用黑白染色法和深度优先遍历判断二分图
G：无向图邻接表
color：顶点颜色，-1/0/1 分别表示未染色/黑色/白色
*/

vector<int> G[MAXN];
int n, color[MAXN];

bool dfs(int u){
    for (int v: G[u]){
        if (color[v] == -1){
            // 未染色节点染成相反颜色
            color[v] = color[u] ^ 1;
            if (!dfs(v)) return false;
        } else if (color[v] == color[u]){
            // 同色边说明存在奇环
            return false;
        }
    }
    return true;
}

bool checkBipartite(){
    fill(color + 1, color + n + 1, -1);
    // 图可能不连通，每个连通分量分别染色
    for (int u = 1; u <= n; u++){
        if (color[u] != -1) continue;
        color[u] = 0;
        if (!dfs(u)) return false;
    }
    return true;
}
```

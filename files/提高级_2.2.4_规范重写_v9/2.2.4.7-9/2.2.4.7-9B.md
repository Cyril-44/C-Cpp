#### 代码示例

```cpp
/*
以下使用 Tarjan 算法求有向图的强连通分量
low[u]：从 u 的 DFS 子树出发，只经过树边和一条特殊边能到达的最小 DFS 序
inStack：顶点是否在 Tarjan 栈中
sccCnt：强连通分量数量，scc[u]：顶点 u 所属分量
stk：存储尚未划分强连通分量的顶点
*/

vector<int> G[MAXN];
int n, dfn[MAXN], low[MAXN], scc[MAXN];
int dfnCnt, sccCnt;
bool inStack[MAXN];

vector<int> stk;

void tarjan(int u){
    dfn[u] = low[u] = ++dfnCnt;
    stk.push_back(u);
    inStack[u] = true;
    for (int v: G[u]){
        if (dfn[v] == 0){
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else if (inStack[v]){
            // 只用仍在栈内的边更新 low
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (low[u] != dfn[u]) return;

    // u 是 SCC 的根，弹出直至 u
    sccCnt++;
    int v = 0;
    do {
        v = stk.back();
        stk.pop_back();
        inStack[v] = false;
        scc[v] = sccCnt;
    } while (v != u);
}

void findScc(){
    fill(dfn + 1, dfn + n + 1, 0);
    fill(inStack + 1, inStack + n + 1, false);
    dfnCnt = sccCnt = 0;
    stk.clear();
    for (int u = 1; u <= n; u++)
        if (dfn[u] == 0) tarjan(u);
}
```

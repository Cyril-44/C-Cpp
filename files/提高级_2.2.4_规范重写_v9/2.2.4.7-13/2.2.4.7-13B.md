#### 代码示例

倍增算法求最近公共祖先：

```cpp
/*
以下实现求 LCA 的倍增算法
dep：顶点深度，anc[u][i]：顶点 u 的 2^i 级祖先
函数 lca 返回两个顶点的最近公共祖先
约定 dep[root]=1，anc[root][0]=0，dep[0]=0
*/

const int MAXLOG = 20;
int n, dep[MAXN], anc[MAXN][MAXLOG];

void init(){
    for (int i = 1; i < MAXLOG; i++)
        for (int u = 1; u <= n; u++)
            anc[u][i] = anc[anc[u][i - 1]][i - 1];
}

int lca(int u, int v){
    if (dep[u] < dep[v]) swap(u, v);
    // 先对齐深度，再同步向上跳转
    for (int i = MAXLOG - 1; i >= 0; i--)
        if (dep[anc[u][i]] >= dep[v]) u = anc[u][i];
    if (u == v) return u;

    for (int i = MAXLOG - 1; i >= 0; i--){
        if (anc[u][i] != anc[v][i]){
            u = anc[u][i];
            v = anc[v][i];
        }
    }
    return anc[u][0];
}
```

Tarjan 离线算法求最近公共祖先：

```cpp
/*
以下实现求 LCA 的 Tarjan 离线算法
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
query：询问邻接表
dsu、sz：并查集父节点和集合大小
ancestor：各并查集当前对应的树上祖先
ans[id]：编号为 id 的询问答案
*/

struct queryNode{
    int v, id;
};

vector<int> T[MAXN];
vector<queryNode> query[MAXN];
int n, dsu[MAXN], sz[MAXN], ancestor[MAXN], ans[MAXM];
bool finished[MAXN];

int findRoot(int x){
    // 路径压缩
    if (dsu[x] == x) return x;
    dsu[x] = findRoot(dsu[x]);
    return dsu[x];
}

void mergeSet(int x, int y, int p){
    x = findRoot(x);
    y = findRoot(y);
    if (x == y) return;
    // 小集合并入大集合，代表元不必是树上的祖先
    if (sz[x] < sz[y]) swap(x, y);
    dsu[y] = x;
    sz[x] += sz[y];
    ancestor[x] = p;
}

void dfsLca(int u, int parent){
    for (int v: T[u]){
        if (v == parent) continue;
        dfsLca(v, u);
        // 子树处理完后并入 u，集合对应的祖先改为 u
        mergeSet(u, v, u);
    }
    finished[u] = true;
    for (queryNode item: query[u]){
        // 代表元未必是树上祖先，需读取 ancestor
        if (finished[item.v]) ans[item.id] = ancestor[findRoot(item.v)];
    }
}

void tarjanLca(int root){
    fill(finished + 1, finished + n + 1, false);
    for (int u = 1; u <= n; u++){
        dsu[u] = ancestor[u] = u;
        sz[u] = 1;
    }
    dfsLca(root, 0);
}
```

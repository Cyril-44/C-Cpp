#### 代码示例

```cpp
/*
以下使用 Tarjan 算法求无向图的点双连通分量
G：无向图邻接表，同一条边的两个方向共用编号 id
low[u]：从 u 的 DFS 子树出发，只经过树边和一条返祖边能到达的最小 DFS 序
bcc[1..bccCnt]：各点双包含的顶点，桥和孤立点不计入点双
belong[u]：0 表示不属于，正数表示只属于该点双，-1 表示属于多个点双
cut[u]：顶点 u 是否为割点
stk：存储当前尚未划分的顶点
*/

struct edge{
    int v, id;
};

vector<edge> G[MAXN];
vector<int> bcc[MAXN];
int n, dfn[MAXN], low[MAXN], belong[MAXN];
int dfnCnt, bccCnt;
bool cut[MAXN];
vector<int> stk;

void tarjan(int u, int faEdge){
    dfn[u] = low[u] = ++dfnCnt;
    stk.push_back(u);
    int child = 0;
    for (edge e: G[u]){
        int v = e.v;
        if (dfn[v] == 0){
            child++;
            tarjan(v, e.id);
            low[u] = min(low[u], low[v]);
            // 非根节点满足一次即可；根节点至少有两个 DFS 子树
            if (low[v] >= dfn[u] && (faEdge != 0 || child > 1)) cut[u] = true;
            if (low[v] != dfn[u]) continue;
            // 等号说明当前点双在 u 处闭合
            ++bccCnt;
            int x = 0;
            do {
                x = stk.back();
                stk.pop_back();
                bcc[bccCnt].push_back(x);
                if (belong[x] == 0) belong[x] = bccCnt;
                else belong[x] = -1;
            } while (x != v);
            bcc[bccCnt].push_back(u);
            if (belong[u] == 0) belong[u] = bccCnt;
            else belong[u] = -1;
        } else if (e.id != faEdge){
            // 按边编号跳过父边，重边仍参与 low 更新
            low[u] = min(low[u], dfn[v]);
        }
    }
    // u 不可能再进入祖先处闭合的点双
    if (low[u] == dfn[u]) stk.pop_back();
}

void findBcc(){
    fill(dfn + 1, dfn + n + 1, 0);
    fill(belong + 1, belong + n + 1, 0);
    fill(cut + 1, cut + n + 1, false);
    for (int i = 1; i <= bccCnt; i++) bcc[i].clear();
    dfnCnt = bccCnt = 0;
    stk.clear();
    for (int u = 1; u <= n; u++){
        if (dfn[u] != 0) continue;
        tarjan(u, 0);
    }
}
```

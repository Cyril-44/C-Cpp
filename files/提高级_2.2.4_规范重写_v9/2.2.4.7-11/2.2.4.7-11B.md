#### 代码示例

两次广度优先遍历求树的直径：

```cpp
/*
函数 diameter 使用两次广度优先遍历求非负权树的直径
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
vis：节点访问标记，dis：起点到各节点的距离
n：节点数量
*/

struct edge{
    int v, w;
};

vector<edge> T[MAXN];
ll dis[MAXN];
bool vis[MAXN];
int n;

int bfs(int start){
    fill(vis + 1, vis + n + 1, false);
    // 从 start 出发，记录距离最远的节点
    queue<int> q;
    q.push(start);
    vis[start] = true;
    dis[start] = 0;
    int far = start;

    while (!q.empty()){
        int u = q.front();
        q.pop();
        for (edge e: T[u]){
            int v = e.v;
            if (vis[v]) continue;
            vis[v] = true;
            dis[v] = dis[u] + ll(e.w);
            if (dis[v] > dis[far]) far = v;
            q.push(v);
        }
    }
    return far;
}

ll diameter(){
    int p = bfs(1);
    p = bfs(p);
    return dis[p];
}
```

树形动态规划求树的直径：

```cpp
/*
函数 diameterDp 使用树形动态规划求树的直径
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
down：从当前节点向子树延伸的最长距离
vis：节点访问标记，ans：树的直径
*/

ll down[MAXN];
ll ans;

void dfs(int u){
    vis[u] = true;
    for (edge e: T[u]){
        int v = e.v;
        if (vis[v]) continue;
        dfs(v);
        // 两条最长子链经过 u 拼成候选直径
        ans = max(ans, down[u] + down[v] + ll(e.w));
        down[u] = max(down[u], down[v] + ll(e.w));
    }
}

ll diameterDp(){
    fill(vis + 1, vis + n + 1, false);
    fill(down + 1, down + n + 1, 0);
    ans = 0;
    dfs(1);
    return ans;
}
```

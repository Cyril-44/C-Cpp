#### 代码示例

树上点差分与子树和统计：

```cpp
/*
函数 addPath 将路径 (s, t) 上所有节点增加 value
函数 collectPoint 汇总差分，所得 pointAns[u] 为节点 u 的增量
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
fa：父节点，pointDiff：节点差分，pointAns：节点增量
*/

vector<int> T[MAXN];
ll pointDiff[MAXN], pointAns[MAXN];
int n, fa[MAXN];

void addPath(int s, int t, ll value){
    int p = lca(s, t);
    // 子树和汇总后，恰使 s 到 t 路径上的每个节点增加 value
    pointDiff[s] += value;
    pointDiff[t] += value;
    pointDiff[p] -= value;
    pointDiff[fa[p]] -= value;
}

void dfsPoint(int u, int parent){
    pointAns[u] = pointDiff[u];
    for (int v: T[u]){
        if (v == parent) continue;
        dfsPoint(v, u);
        // 子树差分向父节点汇总
        pointAns[u] += pointAns[v];
    }
}

void collectPoint(){
    dfsPoint(1, 0);
}
```

树上边差分与子树和统计：

```cpp
/*
函数 addPathEdge 将路径 (s, t) 上所有边增加 value
函数 collectEdge 汇总差分，所得 edgeAns[u] 为边 (fa[u], u) 的增量
T：树的无向邻接表，T[u] 存 u 的所有相邻节点
fa：父节点，edgeDiff：边差分，edgeAns：边增量
*/

ll edgeDiff[MAXN], edgeAns[MAXN];

void addPathEdge(int s, int t, ll value){
    int p = lca(s, t);
    // 两个端点加、LCA 减两次，汇总后只保留路径上的父子边
    edgeDiff[s] += value;
    edgeDiff[t] += value;
    edgeDiff[p] -= ll(2) * value;
}

void dfsEdge(int u, int parent){
    for (int v: T[u]){
        if (v == parent) continue;
        dfsEdge(v, u);
        edgeDiff[u] += edgeDiff[v];
    }
    // 节点 u 的子树和对应父边 (parent, u)
    edgeAns[u] = edgeDiff[u];
}

void collectEdge(){
    dfsEdge(1, 0);
}
```
